## Usage

```tsx
import { typography, fontSize, fontWeight } from '@your-org/polaris-design-library/tokens';

// Use typography styles
const heading = {
  ...typography.headingLg,
};

// Or compose your own
const customText = {
  fontSize: fontSize['400'],
  fontWeight: fontWeight.semibold,
};
```
